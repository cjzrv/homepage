cmake_minimum_required(VERSION 3.20)
project(linecover VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LINECOVER_VIEWER "Build the raylib desktop/WebAssembly viewer" ON)
if(EMSCRIPTEN)
  add_compile_options(-fexceptions)
endif()
add_library(linecover-core src/core.cpp)
target_include_directories(linecover-core PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(linecover-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NOT EMSCRIPTEN)
  add_executable(linecover-cli src/cli.cpp)
  target_link_libraries(linecover-cli PRIVATE linecover-core)
  enable_testing()
  add_executable(linecover-tests tests/core_test.cpp)
  target_link_libraries(linecover-tests PRIVATE linecover-core)
  add_test(NAME geometry-and-metrics COMMAND linecover-tests)
endif()
if(LINECOVER_VIEWER)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  if(EMSCRIPTEN)
    set(PLATFORM Web CACHE STRING "" FORCE)
  endif()
  # raylib 6.0, immutable commit; FetchContent's source override supports offline/local builds.
  FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG dbc56a87da87d973a9c5baa4e7438a9d20121d28 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(raylib)
  if(EMSCRIPTEN)
    # raylib exports ccall through its interface. Merge our runtime exports rather
    # than letting a second EXPORTED_RUNTIME_METHODS setting silently replace them.
    get_target_property(raylib_web_options raylib INTERFACE_LINK_OPTIONS)
    list(FILTER raylib_web_options EXCLUDE REGEX "EXPORTED_RUNTIME_METHODS")
    set_property(TARGET raylib PROPERTY INTERFACE_LINK_OPTIONS "${raylib_web_options}")
  endif()
  add_executable(linecover src/viewer.cpp)
  target_link_libraries(linecover PRIVATE linecover-core raylib)
  if(EMSCRIPTEN)
    set_target_properties(linecover PROPERTIES SUFFIX ".js")
    target_link_options(linecover PRIVATE
      "-sUSE_GLFW=3" "-sALLOW_MEMORY_GROWTH=1" "-sMODULARIZE=1"
      "-sEXPORT_NAME=createLinecover" "-sEXPORT_ES6=1" "-sENVIRONMENT=web"
      "-sDISABLE_EXCEPTION_CATCHING=0" "-sEXIT_RUNTIME=0"
      "-sEXPORTED_RUNTIME_METHODS=['UTF8ToString','HEAPF64','ccall']"
      "-sEXPORTED_FUNCTIONS=['_main','_malloc','_free','_lc_configure','_lc_load','_lc_zigzag','_lc_seek','_lc_step','_lc_play','_lc_rate','_lc_display','_lc_travel','_lc_resize','_lc_state','_lc_export']")
  endif()
endif()
